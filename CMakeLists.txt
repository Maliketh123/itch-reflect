cmake_minimum_required(VERSION 3.16)
project(itch_reflect CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)
add_executable(test_bytes tests/test_bytes.cpp)
target_include_directories(test_bytes PRIVATE src)
add_executable(test_itch tests/test_itch.cpp)
target_include_directories(test_itch PRIVATE src)
add_executable(test_reader tests/test_reader.cpp)
target_include_directories(test_reader PRIVATE src)
add_executable(itch_summary tools/itch_summary.cpp)
target_include_directories(itch_summary PRIVATE src)
